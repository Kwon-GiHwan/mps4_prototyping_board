## MLEK npu CMakeLists region mapping
# Traditionally they have been used as:
#
# Region 0: Weights and biases (and const tensors)
# Region 1: IFM/OFM/Calculation buffers (tensor arena)
# Region 2: Ethos-U'd dedicated cache region (fast cache memory)
#
# NOTE: The above scheme is completely dependent on Vela and could potentially
#       change.
#
# Common definitions:
# For Ethos-U55/U65/U85, region configs are set as:
#   0 or 1 = AXI0 set
#   2 or 3 = AXI1 set
target_compile_definitions(ethosu_core_driver PRIVATE
    NPU_REGIONCFG_2=1   # AXI0=M0
    NPU_REGIONCFG_3=1   # AXI0=M0
    NPU_REGIONCFG_4=1   # AXI0=M0
    NPU_REGIONCFG_5=1   # AXI0=M0
    NPU_REGIONCFG_6=1   # AXI0=M0
    NPU_REGIONCFG_7=1)  # AXI0=M0

# Definitions relevant for all NPUs but depend on memory mode
if (ETHOS_U_NPU_MEMORY_MODE STREQUAL Dedicated_Sram)
    target_compile_definitions(ethosu_core_driver PRIVATE
        NPU_QCONFIG=3       # AXI1=M1
        NPU_REGIONCFG_0=3   # AXI1=M1
        NPU_REGIONCFG_1=3)  # AXI1=M1
elseif (ETHOS_U_NPU_MEMORY_MODE STREQUAL Shared_Sram)
    target_compile_definitions(ethosu_core_driver PRIVATE
        NPU_QCONFIG=3       # AXI1=M1
        NPU_REGIONCFG_0=3   # AXI1=M1
        NPU_REGIONCFG_1=0)  # AXI0=M0
else()
    target_compile_definitions(ethosu_core_driver PRIVATE
        NPU_QCONFIG=0       # AXI0=M0
        NPU_REGIONCFG_0=0   # AXI0=M0
        NPU_REGIONCFG_1=0)  # AXI0=M0
endif()

## driver run_command_stream
                                   uint32_t cms_length,
                                   const uint64_t *base_addr,
                                   int num_base_addr)
{
    assert(num_base_addr <= NPU_REG_BASEP_ARRLEN);

    struct cmd_r cmd;
    struct regioncfg_r rcfg = {0};
    uint64_t qbase          = ethosu_address_remap((uintptr_t)cmd_stream_ptr, -1);
    assert(qbase <= ADDRESS_MASK);
    LOG_DEBUG("QBASE=0x%016llx, QSIZE=%" PRIu32 ", cmd_stream_ptr=%p", qbase, cms_length, cmd_stream_ptr);

    dev->reg->QBASE.word[0] = qbase & 0xffffffff;
    dev->reg->QBASE.word[1] = qbase >> 32;
    dev->reg->QSIZE.word    = cms_length;
    dev->reg->QCONFIG.word  = ethosu_config_select(qbase, -1);

    for (int i = 0; i < num_base_addr; i++)
    {
        uint64_t addr = ethosu_address_remap(base_addr[i], i);
        assert(addr <= ADDRESS_MASK);
        LOG_DEBUG("BASEP%d=0x%016llx", i, addr);
        dev->reg->BASEP[i].word[0] = addr & 0xffffffff;
        dev->reg->BASEP[i].word[1] = addr >> 32;
        rcfg.word |= ethosu_config_select(addr, i) << (i * 2);
    }

    dev->reg->REGIONCFG.word = rcfg.word;

    cmd.word                        = dev->reg->CMD.word & NPU_CMD_PWR_CLK_MASK;
    cmd.transition_to_running_state = 1;

    dev->reg->CMD.word = cmd.word;
    LOG_DEBUG("CMD=0x%08" PRIx32, cmd.word);
}
